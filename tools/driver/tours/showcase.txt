# Showcase screenshots for the README and portfolio: a three-month Viscose
# Intermediate S1 history (tools/driver/specs/showcase/profile.json). The run
# graph section uses VT Controlsphere; the README sections also compare Reload
# and devTS, with a final-fifteen-second fade in their ten most recent runs.
#
#   uv run python scripts/ui_driver.py tour showcase --report
options --fail-on-warnings
start --profile-spec "${REPO}/tools/driver/specs/showcase/profile.json" --setting 'benchmarks/recentIds=["1d24f105-87db-46d2-8c88-66f34c5d4dad"]' --size 1440x900

section scenarios
wait "scenarioGraphPanel > text:VT Controlsphere Novice S5 Hard (*"
# The menu bar holds focus at startup, which highlights File.
eval "forceActiveFocus()" --on scenarioGraphPanel
click "scenarioGraphPanel > seriesVisibilityCheckBox_1"
click "scenarioGraphPanel > seriesVisibilityCheckBox_5"
wait "scenarioGraphPanel > seriesVisibilityCheckBox_5" --prop checked --equals false
click "scenarioHistoryPanel > type:CheckBox & text:Accuracy"
wait "scenarioHistoryPanel > type:CheckBox & text:Accuracy" --prop checked --equals true
mouse move --at 540 280
shot --name scenarios-overview

section scenario-specs
# One shot per scenario-spec's own scenario. The search is set in one edit because typing it
# a key at a time refilters the list per key and the app warns "DelegateModel::cancel: index out
# range". The search field's own text is the first `text:` match, so the row is @1. The search
# is cleared before each shot so the list shows every scenario.
eval "text = 'Polarized Hell Easy 20% Slower'; textEdited()" --on scenarioSearchField
click "scenarioSearchPanel > text:Polarized Hell Easy 20% Slower@1"
wait "scenarioGraphPanel > text:Polarized Hell Easy 20% Slower (*"
eval "text = ''; textEdited()" --on scenarioSearchField
wait "scenarioSearchPanel > text:Floating Heads Timing 400% Larger"

shot --name spec-polarized-hell-easy-20-slower

section run-graph
click "type:MenuBarItem & text:View"
click "text:Graphs"
click graphVisibility_scenarioHistory
click "type:MenuBarItem & text:View"
click "text:Graphs"
click graphVisibility_dailyPlaytime
wait --expr "!visualSettingsManager.playtimeGraphVisible && !visualSettingsManager.scenarioHistoryGraphVisible"

click "scenarioGraphPanel > seriesVisibilityCheckBox_2"
click "scenarioGraphPanel > seriesVisibilityCheckBox_9"

mouse move --at 0 70
shot --name run-graph-story
hover scenarioGraphPanel --offset 928 300 --settle 600 --shot=run-graph-hover

section benchmarks
resize 1800x900
click benchmarksTab
mouse move --at 0 70
wait summaryAttainedRank --prop text --equals Saffron
shot --name benchmark-rows
# A wheel over the grid scrolls the grid, not the page, so the page is positioned directly
# (`set` is the batch variable command): Flick Tech's first row, then the end of the page,
# in a window just tall enough for the rank history and playtime cards.
eval "contentY = 570" --on "benchmarkWorkspaceScroll > type:Flickable@0"
wait "text:Flick Tech"
shot --name benchmark-rows-flick-tech
resize 1440x684
eval "contentY = contentHeight - height" --on "benchmarkWorkspaceScroll > type:Flickable@0"
wait rankHistoryCard
shot --name benchmark-history
resize 1440x900
